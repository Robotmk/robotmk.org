Aufruf: partial "blog_taxonomy_filter.html" (dict "page" . "pages" .Pages) */ -}} {{- $ctx := .page -}} {{- $pages := .pages -}} {{- $topicLimit := 10 -}} {{- $counts := newScratch -}} {{- range $p := $pages -}} {{- range $p.Params.categories -}}{{ $counts.Add (printf "categories|%s" (. | urlize)) 1 }}{{- end -}} {{- range $p.Params.tags -}}{{ $counts.Add (printf "tags|%s" (. | urlize)) 1 }}{{- end -}} {{- end -}} {{- $categories := slice -}} {{- $topics := slice -}} {{- range $key, $count := $counts.Values -}} {{- $parts := split $key "|" -}} {{- $taxo := index $parts 0 -}} {{- $slug := index $parts 1 -}} {{- $term := $ctx.Site.GetPage (printf "/%s/%s" $taxo $slug) -}} {{- $name := $slug | humanize -}} {{- $url := printf "/%s/%s/" $taxo $slug | relLangURL -}} {{- with $term -}} {{- $name = .Title -}} {{- $url = .RelPermalink -}} {{- end -}} {{- $item := dict "key" (printf "%s:%s" $taxo $slug) "name" $name "url" $url "count" $count -}} {{- if eq $taxo "categories" -}} {{- $categories = $categories | append $item -}} {{- else -}} {{- $topics = $topics | append $item -}} {{- end -}} {{- end -}} {{- /* Nach Name vorsortieren, damit gleich häufige Terms alphabetisch stehen. */ -}} {{- $categories = sort (sort $categories "name") "count" "desc" -}} {{- $topics = sort (sort $topics "name") "count" "desc" -}} {{- $topTopics := first $topicLimit $topics -}} {{- $restTopics := after (len $topTopics) $topics -}} {{- if or $categories $topics }}
